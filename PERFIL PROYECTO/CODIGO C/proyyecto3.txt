#include <stdio.h>
#include <string.h>
#include <stdlib.h>

#define MINIMO 10

typedef struct {
    char codigo[20];
    char nombre[20];
    int cantidad;
    double precio;
} Producto;

Producto inventario[100];
int total = 0;

void registrar() {
    if (total >= 100) {
        printf("\n[!] Inventario lleno.\n");
        return;
    }
    printf("\nCodigo (BOT600, BOT1LT, BIDON): ");
    scanf("%s", inventario[total].codigo);
    printf("Nombre: ");
    scanf("%s", inventario[total].nombre);
    printf("Cantidad de botellas que ingresas: ");
    scanf("%d", &inventario[total].cantidad);
    printf("Valor de unidad (precio): ");
    scanf("%lf", &inventario[total].precio);
    total++;
    printf("[+] Guardado con exito!\n");
}

void consultar() {
    printf("\n--- INVENTARIO CON ALERTAS ---\n");
    if (total == 0) {
        printf("Vacio.\n");
        return;
    }
    for (int i = 0; i < total; i++) {
        printf("Cod: %s | %s | Cant: %d | Precio: $%.2f",
               inventario[i].codigo,
               inventario[i].nombre,
               inventario[i].cantidad,
               inventario[i].precio);
        if (inventario[i].cantidad < MINIMO) {
            printf(" <-- [ALERTA: STOCK BAJO]");
        }
        printf("\n");
    }
}

void buscar() {
    char codBuscado[20];
    int encontrado = 0;
    printf("\nIngrese codigo a buscar: ");
    scanf("%s", codBuscado);
    for (int i = 0; i < total; i++) {
        if (strcmp(inventario[i].codigo, codBuscado) == 0) {
            printf("\n[+] Encontrado -> %s | Cant: %d | $%.2f\n",
                   inventario[i].nombre,
                   inventario[i].cantidad,
                   inventario[i].precio);
            encontrado = 1;
            break;
        }
    }
    if (!encontrado) {
        printf("\n[!] No existe.\n");
    }
}

void actualizar() {
    char cod[20];
    printf("\nCodigo del producto a modificar: ");
    scanf("%s", cod);
    for (int i = 0; i < total; i++) {
        if (strcmp(inventario[i].codigo, cod) == 0) {
            int cantidadAnterior = inventario[i].cantidad;
            int nuevaCantidad;
            printf("Producto: %s (Cantidad actual: %d)\n", inventario[i].nombre, inventario[i].cantidad);
            printf("Ingrese la NUEVA cantidad de botellas: ");
            scanf("%d", &nuevaCantidad);

            int diferencia = nuevaCantidad - cantidadAnterior;
            inventario[i].cantidad = nuevaCantidad;

            if (diferencia < 0) {
                int unidadesVendidas = -diferencia;
                double valorVendido = unidadesVendidas * inventario[i].precio;
                printf("[+] Stock actualizado. Se vendieron %d unidades.\n", unidadesVendidas);
                printf("[i] VALOR TOTAL DE LO VENDIDO: $%.2f\n", valorVendido);
            } else if (diferencia > 0) {
                double valorCompra = diferencia * inventario[i].precio;
                printf("[+] Stock actualizado. Ingresaron %d unidades.\n", diferencia);
                printf("[i] Valor total de la compra: $%.2f\n", valorCompra);
            } else {
                printf("[i] No hubo cambios en el stock.\n");
            }
            return;
        }
    }
    printf("[!] No se encontro el producto.\n");
}

void eliminar() {
    char cod[20];
    printf("\nCodigo del producto a eliminar: ");
    scanf("%s", cod);
    for (int i = 0; i < total; i++) {
        if (strcmp(inventario[i].codigo, cod) == 0) {
            for (int j = i; j < total - 1; j++) {
                inventario[j] = inventario[j + 1];
            }
            total--;
            printf("[+] Producto eliminado con exito!\n");
            return;
        }
    }
    printf("[!] No se encontro el producto.\n");
}

void generarInformeExcel() {
    char ruta[300];
    int totalUnidades = 0;
    double valorTotal = 0.0;
    FILE *archivo;

#ifdef _WIN32
    sprintf(ruta, "%s\\Downloads\\informe.csv", getenv("USERPROFILE"));
#else
    sprintf(ruta, "%s/Downloads/informe.csv", getenv("HOME"));
#endif

    archivo = fopen(ruta, "w");
    if (archivo == NULL) {
        printf("\n[!] No se pudo crear el archivo. Cierra el Excel si lo tienes abierto.\n");
        return;
    }

    fprintf(archivo, "sep=,\nINFORME DIARIO DE INVENTARIO - VITALIC\n\n");
    fprintf(archivo, "Codigo,Nombre,Cantidad,Precio Unitario,Subtotal\n");
    for (int i = 0; i < total; i++) {
        totalUnidades += inventario[i].cantidad;
        valorTotal += inventario[i].cantidad * inventario[i].precio;
        fprintf(archivo, "%s,%s,%d,%.2f,%.2f\n", inventario[i].codigo, inventario[i].nombre,
                inventario[i].cantidad, inventario[i].precio,
                inventario[i].cantidad * inventario[i].precio);
    }
    fprintf(archivo, "\nTotal de Productos Distintos,%d\n", total);
    fprintf(archivo, "Total de Unidades,,%d\n", totalUnidades);
    fprintf(archivo, "Valor Total del Stock,,,,%.2f\n", valorTotal);
    fclose(archivo);

    printf("\n[+] Informe exportado a: %s\n", ruta);
}

int main() {
    int opcion;
    do {
        printf("\n=========================================\n");
        printf("       SISTEMA DE INVENTARIO - VITALIC   \n");
        printf("=========================================\n");
        printf("1. Registrar botellas\n");
        printf("2. Consultar Inventario\n");
        printf("3. Buscar producto\n");
        printf("4. Actualizar stock\n");
        printf("5. Eliminar productos\n");
        printf("6. Guardar Informe en Excel\n");
        printf("7. Salir\n");
        printf("=========================================\n");
        printf("Seleccione una opcion: ");
        if (scanf("%d", &opcion) != 1) {
            while (getchar() != '\n');
            printf("\n[!] Error: Por favor, ingrese un numero valido.\n");
            continue;
        }
        getchar();
        switch (opcion) {
            case 1: registrar(); break;
            case 2: consultar(); break;
            case 3: buscar(); break;
            case 4: actualizar(); break;
            case 5: eliminar(); break;
            case 6: generarInformeExcel(); break;
            case 7: printf("\nSaliendo del sistema de inventario VITALIC. Hasta pronto!\n"); break;
            default: printf("\n[!] Opcion no valida. Intente de nuevo.\n");
        }
    } while (opcion != 7);
    return 0;
}
